import pandas as pd
import time
from datetime import datetime
import random

# Define systems and their simulated data metrics
SYSTEMS = {
    "gas_turbine_fleet": {"power_mw": 0},
    "solar_pv": {"output_mw": 0},
    "bess": {"status_percent": 0},
    "compute_racks": {"load_percent": 0},
    "cooling_plant": {"temp_c": 0}
}

# CSV file path
CSV_FILE = "system_stats.csv"

# Initialize CSV with headers
def initialize_csv():
    headers = [
        "timestamp",
        "gas_turbine_power_mw",
        "solar_pv_output_mw",
        "bess_status_percent",
        "compute_racks_load_percent",
        "cooling_plant_temp_c"
    ]
    pd.DataFrame(columns=headers).to_csv(CSV_FILE, index=False)

# Simulate data for each system
def simulate_data():
    SYSTEMS["gas_turbine_fleet"]["power_mw"] = round(random.uniform(450, 550), 2)
    SYSTEMS["solar_pv"]["output_mw"] = round(random.uniform(150, 250), 2)
    SYSTEMS["bess"]["status_percent"] = random.randint(80, 100)
    SYSTEMS["compute_racks"]["load_percent"] = random.randint(70, 95)
    SYSTEMS["cooling_plant"]["temp_c"] = random.randint(20, 30)

# Log data to CSV
def log_data():
    timestamp = datetime.now().strftime("%Y-%m-%d %H:%M:%S")
    data = {
        "timestamp": [timestamp],
        "gas_turbine_power_mw": [SYSTEMS["gas_turbine_fleet"]["power_mw"]],
        "solar_pv_output_mw": [SYSTEMS["solar_pv"]["output_mw"]],
        "bess_status_percent": [SYSTEMS["bess"]["status_percent"]],
        "compute_racks_load_percent": [SYSTEMS["compute_racks"]["load_percent"]],
        "cooling_plant_temp_c": [SYSTEMS["cooling_plant"]["temp_c"]]
    }
    df = pd.DataFrame(data)
    df.to_csv(CSV_FILE, mode="a", header=False, index=False)
    print(f"Logged at {timestamp}: {data}")

# Main loop
def main():
    initialize_csv()
    print(f"Logging data to {CSV_FILE} every 1 second. Press Ctrl+C to stop.")
    try:
        while True:
            simulate_data()
            log_data()
            time.sleep(1)
    except KeyboardInterrupt:
        print("\nStopped logging.")

if __name__ == "__main__":
    main()
