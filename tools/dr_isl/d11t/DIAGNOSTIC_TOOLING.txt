def save(name, value):
    data = json.dumps(value, indent=2, allow_nan=True) + "\n"
    assert len(data.encode()) <= 50_000_000
    temp = D / (name + ".tmp")
    temp.write_text(data)
    temp.replace(D / name)


def pair(case):
    scenario, arm = case
    name = f"{scenario}_{arm}"
    result = dict(scenario=scenario, arm=arm, ticks=0, differences=0,
                  max_public_difference_mw=0.0, changed_public_ticks=[],
                  first_failure=None)
    processes = []
    logs = []
    try:
        for mode in ("stash", "d11t"):
            log = (D / f"{name}.{mode}.stderr.log").open("w")
            logs.append(log)
            processes.append(subprocess.Popen(
                [sys.executable, str(D / "worker.py"), str(ROOT), scenario, arm, "5225a05"],
                stdout=subprocess.PIPE, stderr=log, text=True,
                env=dict(os.environ, D11T_MODE=mode, PYTHONDONTWRITEBYTECODE="1")))
        while True:
            messages = []
            for process in processes:
                line = process.stdout.readline()
                if not line:
                    raise RuntimeError(f"{name}: worker exit {process.poll()}")
                messages.append(json.loads(line))
            if all(m["kind"] == "summary" for m in messages):
                result["summaries"] = dict(zip(("stash", "d11t"), messages))
                break
            if any(m["kind"] != "tick" for m in messages):
                raise RuntimeError("unequal tick counts")
            result["ticks"] += 1
            left, right = messages
            changed = {key: [left["physical"].get(key), right["physical"].get(key)]
                       for key in left["physical"].keys() | right["physical"].keys()
                       if left["physical"].get(key) != right["physical"].get(key)}
            for key in ("frequency_hz", "frequency_valid"):
                if left["frequency"][key] != right["frequency"][key]:
                    changed[key] = [left["frequency"][key], right["frequency"][key]]
            if changed:
                result["first_failure"] = dict(tick=left["frequency"]["tick_index"],
                                                fields=changed)
                break
            a, b = left["frequency"], right["frequency"]
            delta = abs(a["frequency_forcing_mw"] - b["frequency_forcing_mw"])
            if delta:
                result["differences"] += 1
                result["max_public_difference_mw"] = max(
                    result["max_public_difference_mw"], delta)
                result["changed_public_ticks"].append(dict(
                    tick=b["tick_index"], sim_time=b["sim_time"],
                    stash=a["frequency_forcing_mw"], d11t=b["frequency_forcing_mw"],
                    fc_trips=b["fc_trips"], ufls=b["ufls"]))
        if result["first_failure"] is None:
            reference = json.loads(
                (D.parent / "d11" / f"{scenario}_{arm}_1.json").read_text())
            expected = reference["summaries"]["base"]["physical_sha256"]
            result["base_5225a05_nonfrequency_hash"] = expected
            result["nonfrequency_hash_identical_to_base"] = (
                result["summaries"]["d11t"]["physical_sha256"] == expected)
            before = reference["summaries"]["base"]["metrics"]
            after = result["summaries"]["d11t"]["metrics"]
            result["original_metric_differences"] = {
                key: [before.get(key), after.get(key)]
                for key in before.keys() | after.keys()
                if key not in ("invalid_frequency_tick_count", "invalid_frequency_fraction")
                and before.get(key) != after.get(key)}
            result["passed"] = (result["nonfrequency_hash_identical_to_base"]
                                and not result["original_metric_differences"])
        else:
            result["passed"] = False
        save(name + ".json", result)
        return result
    finally:
        for process in processes:
            if process.poll() is None:
                process.terminate()
        for process in processes:
            process.wait()
        for log in logs:
            log.close()


if __name__ == "__main__":
    progress = []
    with ThreadPoolExecutor(max_workers=2) as pool:
        # Four workers maximum; results consumed in deterministic case order.
        for result in pool.map(pair, CASES):
            progress.append({k: v for k, v in result.items() if k != "summaries"})
            save("PROGRESS.json", progress)
            print(json.dumps({k: result[k] for k in (
                "scenario", "arm", "passed", "ticks", "differences",
                "max_public_difference_mw")}), flush=True)
            if not result["passed"]:
                save("STOP.json", result)
                raise SystemExit(2)
    sys.path.insert(0, str(ROOT / "attached_assets/gridsignal_sim_v2_build_package/gridsignal_sim_v2/gridsignal_sim"))
    from study_harness.acceptance import compare_metrics
    verdicts = {}
    for scenario, _ in CASES[:6:2]:
        records = [json.loads((D / f"{scenario}_{arm}.json").read_text())
                   ["summaries"]["d11t"]["record"] for arm in ("baseline", "informed")]
        verdicts[scenario] = compare_metrics(*records)
    save("COMPARE_RUNS.json", verdicts)
    save("STATUS.json", dict(state="T3_COMPLETE", cases=len(progress)))

### suite_gate.py
"""Read completed suite evidence; compare node IDs and known failure assertions."""
import json
from pathlib import Path
import re
import xml.etree.ElementTree as ET

D = Path(__file__).resolve().parent
ROOT = D.parents[2]


def failures(path):
    result = {}
    for c in ET.parse(path).iter("testcase"):
        f = c.find("failure")
        if f is None:
            f = c.find("error")
        if f is not None:
            key = c.attrib["classname"].replace(".", "/") + ".py::" + c.attrib["name"]
            # Test classes are included in JUnit classname, not the source path.
            parts = c.attrib["classname"].split(".")
            index = next((i for i, part in enumerate(parts) if part.startswith("Test")), len(parts))
            key = "/".join(parts[:index]) + ".py"
            if index < len(parts):
                key += "::" + "::".join(parts[index:])
            key += "::" + c.attrib["name"]
            result[key] = f.text or f.get("message", "")
    return result


base_ids = set((ROOT / "tools/dr_isl/PHASE_5_FAILING_TESTS.txt").read_text().splitlines())
actual_ids = set((D / "SUITE_FAILING_TESTS.incremental.txt").read_text().splitlines())
actual = failures(D / "FULL_SUITE_FINAL.xml")
before = failures(ROOT / "tools/dr_isl/phase5/P5_FULL_SUITE_FINAL.xml")
assert actual_ids == set(actual)
gpu = "tests/test_gpu_load_profile.py::test_gpu_load_profile"
allowed_added = {gpu}
signatures = []
for node in sorted(base_ids & actual_ids):
    a, b = before[node], actual[node]
    assertion = re.search(r"(?:^|\n)(?:>?\s*)assert ([^\n]+)", a)
    text = "assert " + assertion.group(1).strip() if assertion else None
    same = bool(text and text in b)
    if not text:
        # Phase 5 records the existing lead-time observer AttributeError.
        text = "'str' object has no attribute 'value'"
        same = text in a and text in b
    signatures.append(dict(nodeid=node, baseline_assertion=text,
                           same_failure_signature=same,
                           base_failure=a, d11t_failure=b))
counts = json.loads((D / "SUITE_COUNTS.json").read_text())
gate = dict(counts=counts, reference_failure_count=len(base_ids),
            added=sorted(actual_ids - base_ids),
            removed=sorted(base_ids - actual_ids),
            remaining=sorted(actual_ids & base_ids),
            gpu_passed=gpu not in actual_ids,
            waived_gpu_only=bool((actual_ids - base_ids) == {gpu}),
            known_signature_count=len(signatures),
            unchanged_known_signatures=sum(s["same_failure_signature"] for s in signatures))
gate["passed"] = (not ((actual_ids - base_ids) - allowed_added)
                  and all(s["same_failure_signature"] for s in signatures))
(D / "KNOWN_FAILURE_SIGNATURES.json").write_text(json.dumps(signatures, indent=2) + "\n")
(D / "FINAL_GATE.json").write_text(json.dumps(gate, indent=2) + "\n")
(ROOT / "tools/dr_isl/D11T_FAILING_TESTS.txt").write_text(
    "".join(node + "\n" for node in sorted(actual_ids)))
print(json.dumps(gate, indent=2))


### suite_observer.py
"""Incremental suite evidence; tests and their ordering are unchanged."""
import json
from pathlib import Path
from P2_observer import *
from P2_observer import pytest_runtest_logreport as observed_report
from P2_observer import pytest_sessionfinish as observed_finish

DEST = Path(__file__).resolve().parent
FAILURES = set()


def pytest_runtest_logreport(report):
    observed_report(report)
    if report.failed:
        FAILURES.add(report.nodeid)
    (DEST / "SUITE_FAILING_TESTS.incremental.txt").write_text(
        "".join(node + "\n" for node in sorted(FAILURES)))


def pytest_sessionfinish(session, exitstatus):
    observed_finish(session, exitstatus)
    reporter = session.config.pluginmanager.getplugin("terminalreporter")
    counts = {name: len(reporter.stats.get(name, [])) for name in (
        "passed", "failed", "skipped", "xfailed", "xpassed", "error")}
    counts["exitstatus"] = int(exitstatus)
    (DEST / "SUITE_COUNTS.json").write_text(json.dumps(counts, indent=2) + "\n")
